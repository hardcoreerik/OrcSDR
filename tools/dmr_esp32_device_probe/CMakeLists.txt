cmake_minimum_required(VERSION 3.16)
project(orcsdr_dmr_esp32_device_probe LANGUAGES CXX)
option(DMR_DEVICE_PROBE_WITH_DECODER "Link actual D0/D1 frontend/protocol/bridge for benchmark" ON)
option(DMR_DEVICE_PROBE_SANITIZERS "Use ASan/UBSan for host tests" OFF)
find_package(Threads REQUIRED)
add_library(dmr_probe_handoff INTERFACE)
target_include_directories(dmr_probe_handoff INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(dmr_probe_handoff INTERFACE cxx_std_17)
add_executable(dmr_probe_handoff_test tests/test_handoff.cpp)
target_link_libraries(dmr_probe_handoff_test PRIVATE dmr_probe_handoff Threads::Threads)
if(MSVC)
  target_compile_options(dmr_probe_handoff_test PRIVATE /W4 /WX)
else()
  target_compile_options(dmr_probe_handoff_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(DMR_DEVICE_PROBE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers need GCC/Clang")
  endif()
  foreach(tgt IN ITEMS dmr_probe_handoff_test)
    target_compile_options(${tgt} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${tgt} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME dmr_probe_handoff COMMAND dmr_probe_handoff_test)
if(DMR_DEVICE_PROBE_WITH_DECODER)
  set(DMR "${CMAKE_CURRENT_SOURCE_DIR}/../../components/orcsdr_dmr")
  add_library(dmr_probe_native STATIC
    "${DMR}/src/d0.cpp"
    "${DMR}/d1_protocol/src/d1.cpp"
    "${DMR}/d1_frontend/src/frontend.cpp"
    "${DMR}/d1_integration/src/d1_bridge.cpp")
  target_include_directories(dmr_probe_native PUBLIC
    "${DMR}/include" "${DMR}/d1_protocol/include"
    "${DMR}/d1_frontend/include" "${DMR}/d1_integration/include")
  target_compile_features(dmr_probe_native PUBLIC cxx_std_17)
  add_executable(dmr_probe_host src/probe_host.cpp src/pattern.cpp)
  target_link_libraries(dmr_probe_host PRIVATE dmr_probe_native dmr_probe_handoff)
  if(MSVC)
    target_compile_options(dmr_probe_native PRIVATE /W4 /WX)
    target_compile_options(dmr_probe_host PRIVATE /W4 /WX)
  else()
    target_compile_options(dmr_probe_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(dmr_probe_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(DMR_DEVICE_PROBE_SANITIZERS)
    foreach(tgt IN ITEMS dmr_probe_native dmr_probe_host)
      target_compile_options(${tgt} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${tgt} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
  add_test(NAME dmr_probe_native_smoke COMMAND dmr_probe_host --seconds 1 --no-log)
endif()
