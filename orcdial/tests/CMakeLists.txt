cmake_minimum_required(VERSION 3.20)
project(orcdial_security_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT MBEDTLS_SOURCE_DIR)
  message(FATAL_ERROR "Pass -DMBEDTLS_SOURCE_DIR=<mbedtls source tree>")
endif()
file(STRINGS "${MBEDTLS_SOURCE_DIR}/include/mbedtls/build_info.h" mbedtls_major REGEX "^#define MBEDTLS_VERSION_MAJOR +[0-9]+")
file(STRINGS "${MBEDTLS_SOURCE_DIR}/include/mbedtls/build_info.h" mbedtls_minor REGEX "^#define MBEDTLS_VERSION_MINOR +[0-9]+")
if(NOT mbedtls_major MATCHES "MAJOR +3$" OR NOT mbedtls_minor MATCHES "MINOR +6$")
  message(FATAL_ERROR "Host security tests require Mbed TLS 3.6 LTS sources (validated with 3.6.5).")
endif()
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE_DIR}" mbedtls EXCLUDE_FROM_ALL)
add_executable(security_test security_test.cpp)
target_link_libraries(security_test PRIVATE mbedcrypto)
add_executable(protocol_test protocol_test.cpp)
add_executable(controller_test controller_test.cpp)
add_executable(runtime_test runtime_test.cpp)
add_executable(policy_test policy_test.cpp)
target_include_directories(runtime_test PRIVATE runtime_stubs)
target_link_libraries(runtime_test PRIVATE mbedcrypto)
foreach(test_target security_test protocol_test controller_test runtime_test policy_test)
  if(MSVC)
    target_compile_options(${test_target} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endif()
endforeach()
enable_testing()
add_test(NAME runtime_failure COMMAND runtime_test)
add_test(NAME policy COMMAND policy_test)
add_test(NAME security COMMAND security_test)
add_test(NAME protocol COMMAND protocol_test)
add_test(NAME controller COMMAND controller_test)
