cmake_minimum_required(VERSION 3.16)
project(orc_zero_airband LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(orc_zero_airband STATIC airband.cpp airband_render.cpp)
target_include_directories(orc_zero_airband PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(orc_zero_airband PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(orc_zero_airband_preview demo.cpp)
target_link_libraries(orc_zero_airband_preview PRIVATE orc_zero_airband)
include(CTest)
if(BUILD_TESTING)
  add_executable(orc_zero_airband_tests tests/airband_tests.cpp)
  target_link_libraries(orc_zero_airband_tests PRIVATE orc_zero_airband)
  add_test(NAME airband_regression COMMAND orc_zero_airband_tests)
endif()
