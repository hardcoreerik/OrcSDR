cmake_minimum_required(VERSION 3.16)
project(orcui_host_tests LANGUAGES CXX)
add_executable(orcui_layout_test ../layout.cpp ../raster.cpp layout_test.cpp)
target_include_directories(orcui_layout_test PRIVATE ../include)
target_compile_features(orcui_layout_test PRIVATE cxx_std_17)
target_compile_options(orcui_layout_test PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(ORCUI_LVGL_SOURCE)
  enable_language(C)
  set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${ORCUI_LVGL_SOURCE}" "${CMAKE_BINARY_DIR}/lvgl")
  add_executable(orcui_lvgl_test lvgl_test.cpp ../home_lvgl.cpp ../layout.cpp ../raster.cpp)
  target_include_directories(orcui_lvgl_test PRIVATE ../include ../../../apps/orcsdr-waveshare-4b/main)
  target_compile_features(orcui_lvgl_test PRIVATE cxx_std_17)
  target_compile_definitions(orcui_lvgl_test PRIVATE ORCUI_BADGE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../../../apps/orcsdr-waveshare-4b/main/orc_badge_104.rgb565")
  target_compile_options(orcui_lvgl_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(orcui_lvgl_test PRIVATE lvgl)
  add_test(NAME orcui_lvgl COMMAND orcui_lvgl_test)
