cmake_minimum_required(VERSION 3.21)
project(orcui_desktop LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build LVGL statically" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "LVGL demos" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "LVGL examples" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "ThorVG" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "LVGL config" FORCE)
set(ORCUI_LVGL_SOURCE "" CACHE PATH "Existing LVGL v9.3.0 checkout for offline builds")
include(FetchContent)
if(ORCUI_LVGL_SOURCE)
  if(NOT EXISTS "${ORCUI_LVGL_SOURCE}/lvgl.h")
    message(FATAL_ERROR "ORCUI_LVGL_SOURCE must contain lvgl.h")
  endif()
  add_subdirectory("${ORCUI_LVGL_SOURCE}" "${CMAKE_BINARY_DIR}/lvgl")
else()
  FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git
    GIT_TAG v9.3.0 GIT_SHALLOW TRUE GIT_PROGRESS TRUE)
  FetchContent_MakeAvailable(lvgl)
endif()
find_package(SDL2 CONFIG QUIET)
if(TARGET SDL2::SDL2)
  set(ORCUI_SDL SDL2::SDL2)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  set(ORCUI_SDL PkgConfig::SDL2)
endif()
target_link_libraries(lvgl PUBLIC ${ORCUI_SDL})
add_executable(orcui_desktop main.cpp ../home_lvgl.cpp ../layout.cpp ../raster.cpp)
target_include_directories(orcui_desktop PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../include")
target_link_libraries(orcui_desktop PRIVATE lvgl ${ORCUI_SDL})
if(MSVC)
  target_compile_options(orcui_desktop PRIVATE /W4)
else()
  target_compile_options(orcui_desktop PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME desktop_sdl_smoke COMMAND orcui_desktop --preset waveshare --seconds 2)
  set_tests_properties(desktop_sdl_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 15)
endif()
