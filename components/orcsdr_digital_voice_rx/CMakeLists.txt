cmake_minimum_required(VERSION 3.16)
# ESP-IDF auto-discovers every directory under components/. These are host-only
# today (their suites build standalone); register an empty component so the
# firmware configure step succeeds. Sources are added when firmware wiring lands.
if(COMMAND idf_component_register)
    idf_component_register()
    return()
endif()
project(orcsdr_digital_voice_rx_host LANGUAGES CXX)
option(ORC_DV_SANITIZERS "Compile host-only receive protocol gates with ASan+UBSan" OFF)
add_library(orcsdr_digital_voice_rx STATIC src/metadata.cpp)
target_compile_features(orcsdr_digital_voice_rx PUBLIC cxx_std_17)
target_include_directories(orcsdr_digital_voice_rx PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(orcsdr_dv_rx_tests tests/test_metadata.cpp)
target_link_libraries(orcsdr_dv_rx_tests PRIVATE orcsdr_digital_voice_rx)
add_executable(orcsdr_dv_rx_cli tools/dv_rx_cli.cpp)
target_link_libraries(orcsdr_dv_rx_cli PRIVATE orcsdr_digital_voice_rx)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(t IN ITEMS orcsdr_digital_voice_rx orcsdr_dv_rx_tests orcsdr_dv_rx_cli)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ORC_DV_SANITIZERS)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
enable_testing()
add_test(NAME orcsdr_dv_metadata_fec_crc COMMAND orcsdr_dv_rx_tests)
