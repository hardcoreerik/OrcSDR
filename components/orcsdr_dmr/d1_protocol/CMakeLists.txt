cmake_minimum_required(VERSION 3.16)
project(orcsdr_dmr_d1 LANGUAGES CXX)

# Standalone host-only build; owns no changes to parent D0 or firmware CMake.
add_library(orcsdr_dmr_d1 STATIC src/d1.cpp ../src/d0.cpp)
target_include_directories(orcsdr_dmr_d1 PUBLIC include ../include)
target_compile_features(orcsdr_dmr_d1 PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(orcsdr_dmr_d1 PRIVATE /W4 /WX)
else()
  target_compile_options(orcsdr_dmr_d1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(ORCSDR_D1_SANITIZERS "Build host tests with address/undefined sanitizers" OFF)
if(ORCSDR_D1_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
  endif()
  target_compile_options(orcsdr_dmr_d1 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(orcsdr_dmr_d1 PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(orcsdr_dmr_d1_test tests/test_d1.cpp)
target_link_libraries(orcsdr_dmr_d1_test PRIVATE orcsdr_dmr_d1)
if(MSVC)
  target_compile_options(orcsdr_dmr_d1_test PRIVATE /W4 /WX)
else()
  target_compile_options(orcsdr_dmr_d1_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME dmr_d1_protocol COMMAND orcsdr_dmr_d1_test)

# D1-S CACH evidence model and end-to-end hard-dibit stream regressions.
# Changes remain exclusively in this owned module; D0 is still read-only.
add_executable(orcsdr_dmr_d1_cach_tracking_test tests/test_cach_tracking.cpp)
target_include_directories(orcsdr_dmr_d1_cach_tracking_test PRIVATE include)
add_executable(orcsdr_dmr_d1_cach_stream_test tests/test_cach_stream.cpp)
target_link_libraries(orcsdr_dmr_d1_cach_stream_test PRIVATE orcsdr_dmr_d1)
foreach(tgt IN ITEMS orcsdr_dmr_d1_cach_tracking_test orcsdr_dmr_d1_cach_stream_test)
  target_compile_features(${tgt} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(${tgt} PRIVATE /W4 /WX)
  else()
    target_compile_options(${tgt} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(ORCSDR_D1_SANITIZERS)
  target_compile_options(orcsdr_dmr_d1_cach_tracking_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(orcsdr_dmr_d1_cach_tracking_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME dmr_d1_cach_tracking COMMAND orcsdr_dmr_d1_cach_tracking_test)
add_test(NAME dmr_d1_cach_stream COMMAND orcsdr_dmr_d1_cach_stream_test)
