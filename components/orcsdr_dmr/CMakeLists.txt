cmake_minimum_required(VERSION 3.16)
project(orcsdr_dmr_d0 LANGUAGES CXX)

# Deliberately host-only. Not registered with ESP-IDF or linked into firmware.
add_library(orcsdr_dmr_d0 STATIC src/d0.cpp)
target_include_directories(orcsdr_dmr_d0 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(orcsdr_dmr_d0 PUBLIC cxx_std_17)

option(ORCSDR_DMR_STRICT "Treat compiler warnings as errors" ON)
option(ORCSDR_DMR_SANITIZE "Use ASan + UBSan for host tests" OFF)
if(ORCSDR_DMR_STRICT)
    if(MSVC)
        target_compile_options(orcsdr_dmr_d0 PRIVATE /W4 /WX)
    else()
        target_compile_options(orcsdr_dmr_d0 PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
if(ORCSDR_DMR_SANITIZE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizers require GCC or Clang")
    endif()
    target_compile_options(orcsdr_dmr_d0 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(orcsdr_dmr_d0 PUBLIC -fsanitize=address,undefined)
endif()

enable_testing()
foreach(suite IN ITEMS sync cach bptc)
    add_executable(orcsdr_dmr_test_${suite} tests/test_${suite}.cpp)
    target_link_libraries(orcsdr_dmr_test_${suite} PRIVATE orcsdr_dmr_d0)
    if(ORCSDR_DMR_STRICT)
        if(MSVC)
            target_compile_options(orcsdr_dmr_test_${suite} PRIVATE /W4 /WX)
        else()
            target_compile_options(orcsdr_dmr_test_${suite} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
    add_test(NAME dmr_d0_${suite} COMMAND orcsdr_dmr_test_${suite})
endforeach()
