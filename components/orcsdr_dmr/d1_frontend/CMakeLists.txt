cmake_minimum_required(VERSION 3.16)
project(orcsdr_dmr_d1_frontend LANGUAGES CXX)
# Standalone host target: does not alter the parent D0 or ESP-IDF build.
add_library(orcsdr_dmr_d1 STATIC src/frontend.cpp)
target_include_directories(orcsdr_dmr_d1 PUBLIC include PRIVATE ../include)
target_compile_features(orcsdr_dmr_d1 PUBLIC cxx_std_17)
option(DMR_D1_SANITIZE "Enable ASan and UBSan on host" OFF)
if(MSVC)
  target_compile_options(orcsdr_dmr_d1 PRIVATE /W4 /WX)
else()
  target_compile_options(orcsdr_dmr_d1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(DMR_D1_SANITIZE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
  endif()
  target_compile_options(orcsdr_dmr_d1 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(orcsdr_dmr_d1 PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(dmr_d1_host_tests tests/test_frontend.cpp)
target_link_libraries(dmr_d1_host_tests PRIVATE orcsdr_dmr_d1)
target_include_directories(dmr_d1_host_tests PRIVATE ../include)
if(MSVC)
  target_compile_options(dmr_d1_host_tests PRIVATE /W4 /WX)
else()
  target_compile_options(dmr_d1_host_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME dmr_d1_frontend COMMAND dmr_d1_host_tests)
