cmake_minimum_required(VERSION 3.16)
# ESP-IDF auto-discovers every directory under components/. These are host-only
# today (their suites build standalone); register an empty component so the
# firmware configure step succeeds. Sources are added when firmware wiring lands.
if(COMMAND idf_component_register)
    idf_component_register()
    return()
endif()
project(orcsdr_nxdn_host LANGUAGES CXX)
option(ORC_NXDN_SANITIZERS "Host NXDN ASan/UBSan (GCC/Clang)" OFF)
add_library(orcsdr_nxdn STATIC src/decoder.cpp src/frontend.cpp)
target_include_directories(orcsdr_nxdn PUBLIC include)
target_compile_features(orcsdr_nxdn PUBLIC cxx_std_17)
add_executable(orcsdr_nxdn_replay tools/replay.cpp)
target_link_libraries(orcsdr_nxdn_replay PRIVATE orcsdr_nxdn)
add_executable(orcsdr_nxdn_tests tests/test_decoder.cpp)
target_link_libraries(orcsdr_nxdn_tests PRIVATE orcsdr_nxdn)
foreach(t IN ITEMS orcsdr_nxdn orcsdr_nxdn_tests orcsdr_nxdn_replay)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /WX)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ORC_NXDN_SANITIZERS)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME nxdn_host_metadata COMMAND orcsdr_nxdn_tests)
