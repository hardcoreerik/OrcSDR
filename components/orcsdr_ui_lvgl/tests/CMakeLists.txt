cmake_minimum_required(VERSION 3.16)
project(orcsdr_ui_host_tests LANGUAGES CXX)
get_filename_component(TAB5_UI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../apps/orcsdr-tab5/ui" ABSOLUTE)
add_executable(home_layout_test home_layout_test.cpp ../src/home_layout.cpp)
target_include_directories(home_layout_test PRIVATE ../include)
target_compile_features(home_layout_test PRIVATE cxx_std_17)
target_compile_definitions(home_layout_test PRIVATE TAB5_UI_DIR="${TAB5_UI_DIR}")
target_compile_options(home_layout_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME home_layout COMMAND home_layout_test)

add_executable(home_plots_test home_plots_test.cpp ../src/home_plots.cpp)
target_include_directories(home_plots_test PRIVATE ../include ../src ${TAB5_UI_DIR})
target_compile_features(home_plots_test PRIVATE cxx_std_17)
target_compile_options(home_plots_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
add_test(NAME home_plots COMMAND home_plots_test)
