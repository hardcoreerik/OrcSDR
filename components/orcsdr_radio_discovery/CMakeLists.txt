cmake_minimum_required(VERSION 3.16)

if(COMMAND idf_component_register)
    idf_component_register(INCLUDE_DIRS "include")
else()
    project(orcsdr_radio_discovery LANGUAGES CXX)
    add_library(orcsdr_radio_discovery INTERFACE)
    target_include_directories(orcsdr_radio_discovery INTERFACE "${CMAKE_CURRENT_LIST_DIR}/include")
    target_compile_features(orcsdr_radio_discovery INTERFACE cxx_std_17)
    enable_testing()
    add_executable(radio_discovery_evidence_tests tests/evidence_tests.cpp)
    target_link_libraries(radio_discovery_evidence_tests PRIVATE orcsdr_radio_discovery)
    add_test(NAME radio_discovery_evidence_tests COMMAND radio_discovery_evidence_tests)
endif()
