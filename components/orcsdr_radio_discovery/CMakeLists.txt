cmake_minimum_required(VERSION 3.16)
if(COMMAND idf_component_register)
    idf_component_register(INCLUDE_DIRS "include")
else()
    project(orcsdr_radio_discovery LANGUAGES CXX)
    add_library(orcsdr_radio_discovery INTERFACE)
    target_include_directories(orcsdr_radio_discovery INTERFACE "${CMAKE_CURRENT_LIST_DIR}/include")
    target_compile_features(orcsdr_radio_discovery INTERFACE cxx_std_17)
    enable_testing()
    foreach(test IN ITEMS evidence storage)
        add_executable(radio_discovery_${test}_tests tests/${test}_tests.cpp)
        target_link_libraries(radio_discovery_${test}_tests PRIVATE orcsdr_radio_discovery)
        if(MSVC)
            target_compile_options(radio_discovery_${test}_tests PRIVATE /W4 /WX)
        else()
            target_compile_options(radio_discovery_${test}_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
        add_test(NAME radio_discovery_${test}_tests COMMAND radio_discovery_${test}_tests)
    endforeach()
endif()
