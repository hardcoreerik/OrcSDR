cmake_minimum_required(VERSION 3.16)
project(orcsdr_digital_voice_ui_host LANGUAGES CXX)
option(ORC_DV_UI_SANITIZERS "Enable ASan/UBSan on host" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(UI ${CMAKE_CURRENT_LIST_DIR}/../../apps/orcsdr-tab5/ui)
# This suite uses command-recording M5Unified stand-ins; not native pixel parity.
set(SOURCES ${UI}/digital_voice_dashboard.cpp
            ${UI}/digital_voice_dashboard_model.cpp
            ${UI}/digital_voice_dashboard_demo.cpp)
foreach(test_target IN ITEMS dv_dashboard_host_contract dv_dashboard_surface_contract)
  if(test_target STREQUAL dv_dashboard_host_contract)
    set(TEST_SOURCE ${CMAKE_CURRENT_LIST_DIR}/dashboard_host_contract_tests.cpp)
  else()
    set(TEST_SOURCE ${CMAKE_CURRENT_LIST_DIR}/dashboard_surface_host_tests.cpp)
  endif()
  add_executable(${test_target} ${TEST_SOURCE} ${SOURCES})
  if(test_target STREQUAL dv_dashboard_surface_contract)
    target_sources(${test_target} PRIVATE ${UI}/digital_voice_dashboard_surface.cpp)
  endif()
  target_include_directories(${test_target} PRIVATE ${CMAKE_CURRENT_LIST_DIR}/mock ${UI})
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ORC_DV_UI_SANITIZERS)
      target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${test_target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME digital_voice_pages COMMAND dv_dashboard_host_contract)
add_test(NAME digital_voice_surface_contract COMMAND dv_dashboard_surface_contract)
